```markdown
### 2.9 同音听写错误：按整篇语义改字

1. 依据是**整篇语义**：前文、后文、事件逻辑都算。只改「原字在这句里读不通、且整篇只有这一种读法成立」的字；不设处数上限。

1. **口水清零**：`呃 / 嗯 / 啊 / 唉 / 哎 / 诶 / 那个 / 就是说 / 然后然后 / 就是就是 / 这个这个 / 对对对 / 反正 / 其实吧 / 怎么说呢` 一律不出现。
2. **元叙述清零**：`我思考一下 / 让我想想 / 就是什么呢 / 我在想 / 怎么说呢 / 你知道吧 / 反正就是` 全删。
3. **不压缩**：没有字数目标。只做去原地重复、合并同义反复；处理完多少字就多少字。
4. **可补 1–2 句解释**：只把原文隐含的因果/感受说明白（例：`特别激动，因为……是一种非常新奇的体验`）；不许新增事件、人物、物件、地点。
5. **主语与指代补全**：`纠结`→`大家一起纠结`、`攻略没做好`→`大家对攻略也没做好`、`店门口说`→`店门口的人说`、`那里吃`→`到了澳门之后在那里吃`；二选一且原文无依据时保持原样。
6. **数字**：只改明显不可能/自相矛盾的（`4.4点钟`→`4点钟`）；合理小数与口语数字不动。
7. **不确定词**：叠字合并、冗余可删，但每个不确定的事实至少留一个模糊词；禁止把推测写成断言。
8. **保留口吻**：第一人称、口语节奏、结尾「非常简短的一个梦」照留。
```

```python
def gate_g5(inp: str, out: str) -> GateResult:
    if not any(w in inp for w in UNCERTAIN):
        return GateResult("G5", True, 0.0, [])
    if not any(w in out for w in UNCERTAIN):
        return GateResult("G5", False, 1.0, ["把不确定的事写成了确定"])
    in_paras = [p for p in inp.split("\n") if p.strip()]
    out_paras = [p for p in out.split("\n") if p.strip()]
    if len(in_paras) == len(out_paras):
        for i, (a, b) in enumerate(zip(in_paras, out_paras)):
            if any(w in a for w in UNCERTAIN) and not any(w in b for w in UNCERTAIN):
                return GateResult("G5", False, float(i + 1), [f"第 {i+1} 段把推测写成了确定"])
    return GateResult("G5", True, 0.0, [])
```

```kotlin
private fun gateF2(input: String, output: String, missingAnchors: Int): GateResult {
    val a = contentCharCount(input)
    val b = contentCharCount(output)
    val ratio = if (a == 0) 0.0 else b.toDouble() / a
    val problems = buildList {
        if (a > 0 && ratio < 0.60 && missingAnchors > 0) {
```

```kotlin
private fun gateG5(input: String, output: String): GateResult {
    val hasIn = UNCERTAIN.any { it in input }
    if (!hasIn) return GateResult(Gate.G5, true, 0.0, emptyList())
    if (UNCERTAIN.none { it in output }) {
        return GateResult(Gate.G5, false, 1.0, listOf("把不确定的事写成了确定"))
    }
    val inParas = input.split("\n").filter { it.isNotBlank() }
    val outParas = output.split("\n").filter { it.isNotBlank() }
    if (inParas.size == outParas.size) {
        inParas.forEachIndexed { i, a ->
            val b = outParas[i]
            if (UNCERTAIN.any { it in a } && UNCERTAIN.none { it in b }) {
                return GateResult(Gate.G5, false, (i + 1).toDouble(), listOf("第 ${i + 1} 段把推测写成了确定"))
            }
        }
    }
    return GateResult(Gate.G5, true, 0.0, emptyList())
}
```

```kotlin
/** 仅 debug：列出输出比输入多出来的字（= 同音改动/新增字），供排查过度改字。 */
internal fun homophoneAudit(input: String, output: String): String {
    val inCounts = FaithfulGates.contentChars(input).groupingBy { it }.eachCount()
    val surplus = FaithfulGates.contentChars(output).groupingBy { it }.eachCount()
        .flatMap { (ch, c) -> List((c - (inCounts[ch] ?: 0)).coerceAtLeast(0)) { ch } }
    return if (surplus.isEmpty()) "多出字=无" else "多出字=" + surplus.take(8).joinToString("")
}
```
