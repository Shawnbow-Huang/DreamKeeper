# 小说化 v3 实施计划（0.13.0）

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 把「把原样稿改写成小说」从一次调用升级成**规划 → 写作 → 逐行润色**三段，并让成品真的像一个短篇：有前提、有主导变化、有语气，而不是把原样稿加几个形容词。

**Architecture:** 三段各一次 HTTP，各自可重试一次；PLAN 产出 JSON（前提/反前提/主题句/3–5 场景/endingChange/coreImages），DRAFT 注入 story-skills 的 scene/prose/anti-slop 与 PLAN，POLISH 只做 line-edit。闸门 N1（结构，真闸门）/ N2（文风，真闸门）/ N3（AI 腔，只提示）。创作手记替代「补写 N 处」。

**Tech Stack:** Kotlin + Compose + Room；Python 3 校验脚本；assets 内嵌 `story-skills`（MIT，131 篇，797KB）；JUnit4 + Robolectric。

## 2. 契约

```
阶段：PLAN → DRAFT → POLISH（每段最多 2 次尝试；PLAN 失败整次失败，DRAFT 失败整次失败，POLISH 失败回退 DRAFT 文本）
PLAN JSON：
{
  "premise": "一句「谁想要什么，但被什么挡住」",
  "counterPremise": "如果反着写会怎样（一句）",
  "theme": "一句话，宁可具体不要大词",
  "scenes": [{"goal":"","turn":"","value":"正→负 或 负→正"}],
  "endingChange": "结尾相对开头改变了什么（一句）",
  "coreImages": ["必须来自原文的意象，逐字子串"]
}
约束：scenes 3–5 条；coreImages 1–3 条且每条都是原文子串；endingChange 非空。
N1：scenes 3–5、endingChange 非空、coreImages ⊆ 原文；全过才允许 DRAFT。
N2：字数 800–5000；filter words ≤10/千字；「地」副词 ≤12/千字；说书标签（话说/且说/看官）≤2/篇；破折号 ≤3/篇；A7 违禁词 = 0。
N3（仅提示）：humanizer 清单命中数，写进证据，不拦。
promptHash = sha256(plan‖draft‖polish‖MANIFEST.json) 前 8 位，写进 novel_versions.promptHash。
创作手记 JSON：[{"type":"借用|舍弃|新造|主题","detail":"..."}]；旧行（[{"type":"补写","quote":...}]）按 quote 兼容读取。
```

```kotlin
enum class SkillStage { PLAN, DRAFT, POLISH }

class SkillLibrary(private val assets: AssetManager) {
    private val manifest by lazy { readManifest() }

    /** 按 stage 的固定清单取正文；顺序固定，保证 promptHash 可复现。 */
    fun filesFor(stage: SkillStage): List<SkillFile> = when (stage) {
        SkillStage.PLAN -> pick("premise-workshop.md", "controlling-idea.md", "short-story-form.md")
        SkillStage.DRAFT -> pick("scene-sequel.md", "prose-checks.md", "anti-slop.md")
        SkillStage.POLISH -> pick("line-edit-checklist.md", "prose-checks.md", "anti-slop.md")
    }.map { name ->
        val bytes = assets.open("skills/story/$name").use { it.readBytes() }
        SkillFile(name, String(bytes, Charsets.UTF_8), bytes.size)
    }

    /** 组合 hash：三段 + MANIFEST，前 8 位。 */
    fun bundleSha8(): String {
        val h = MessageDigest.getInstance("SHA-256")
        val parts = listOf("novel.v3-plan.md", "novel.v3-draft.md", "novel.v3-polish.md")
        parts.forEach { n ->
            h.update(assets.open("prompts/$n").use { it.readBytes() })
        }
        h.update(assets.open("skills/story/MANIFEST.json").use { it.readBytes() })
        return h.digest().joinToString("") { "%02x".format(it) }.take(8)
    }
}
```

`pick` 找不到文件时抛 `IllegalStateException("skill 资产缺失：$name")`，不要静默跳过。

class NovelPrompts(private val context: Context) {
    fun bundleSha8(): String {
        val h = MessageDigest.getInstance("SHA-256")
        listOf(PLAN, DRAFT, POLISH).forEach { spec ->
            val src = PromptRepository(AssetPromptSource(context)).raw(spec.promptId)
            h.update(src.toByteArray(StandardCharsets.UTF_8))
        }
        h.update(SkillLibrary(context.assets).manifestBytes())
        return h.digest().joinToString("") { "%02x".format(it) }.take(8)
    }
}
```

```kotlin
data class NovelScene(val goal: String, val turn: String, val value: String)

data class NovelPlan(
    val premise: String,
    val counterPremise: String,
    val theme: String,
    val scenes: List<NovelScene>,
    val endingChange: String,
    val coreImages: List<String>,
) {
    companion object {
        fun parse(json: String, raw: String): NovelPlan? = runCatching {
            val o = JSONObject(stripFence(json))
            val scenes = o.getJSONArray("scenes").let { arr ->
                (0 until arr.length()).map { i ->
                    arr.getJSONObject(i).let { NovelScene(it.getString("goal"), it.getString("turn"), it.getString("value")) }
                }
            }
            val images = o.getJSONArray("coreImages").let { arr -> (0 until arr.length()).map { arr.getString(it) } }
            NovelPlan(
                premise = o.getString("premise").trim(),
                counterPremise = o.optString("counterPremise").trim(),
                theme = o.getString("theme").trim(),
                scenes = scenes,
                endingChange = o.getString("endingChange").trim(),
                coreImages = images,
            )
        }.getOrNull()?.takeIf { it.isValid(raw) }

        fun stripFence(s: String): String = s.trim().removePrefix("```json").removePrefix("```").removeSuffix("```").trim()
    }

    fun isValid(raw: String): Boolean =
        premise.isNotBlank() && theme.isNotBlank() && endingChange.isNotBlank() &&
            scenes.size in 3..5 && scenes.all { it.goal.isNotBlank() && it.turn.isNotBlank() } &&
            coreImages.size in 1..3 && coreImages.all { raw.contains(it) }
}
```

```python
FILTER_WORDS = ["突然","忽然","仿佛","似乎","好像","只见","顿时","瞬间","不禁","微微","缓缓","淡淡","默默","静静","轻轻","深深"]
TELL_TAGS = ["话说","且说","看官","列位","欲知后事"]
EM_DASH = "—"

def gate_n1(raw, plan):
    """结构：3–5 场景、endingChange 非空、coreImages 都是原文子串。"""
    problems = []
    scenes = plan.get("scenes") or []
    if not (3 <= len(scenes) <= 5):
        problems.append(f"场景数 {len(scenes)} 不在 3–5")
    if not (plan.get("endingChange") or "").strip():
        problems.append("缺 endingChange")
    images = plan.get("coreImages") or []
    if not images:
        problems.append("缺 coreImages")
    for img in images:
        if img not in raw:
            problems.append(f"意象不是原文子串：{img}")
    return ok() if not problems else fail("N1", "；".join(problems))

def gate_n2(text):
    """文风：字数 800–5000；filter words ≤10/千字；地副词 ≤12/千字；说书标签 ≤2；破折号 ≤3。"""
    n = len(content_chars(text))
    problems = []
    if not (800 <= n <= 5000):
        problems.append(f"字数 {n} 不在 800–5000")
    per_k = lambda c: (c * 1000.0 / n) if n else 0.0
    fw = sum(text.count(w) for w in FILTER_WORDS)
    if per_k(fw) > 10:
        problems.append(f"filter words {per_k(fw):.1f}/千字")
    de = len(re.findall(r"[一-鿿]地", text))
    if per_k(de) > 12:
        problems.append(f"「地」副词 {per_k(de):.1f}/千字")
    tags = sum(text.count(w) for w in TELL_TAGS)
    if tags > 2:
        problems.append(f"说书标签 {tags} 次")
    dashes = text.count("—")
    if dashes > 3:
        problems.append(f"破折号 {dashes} 个")
    return ok() if not problems else fail("N2", "；".join(problems))

def gate_n3(text):
    """AI 腔：只返回 hints，不计 failed。"""
    hints = [w for w in HUMANIZER_WORDS if w in text]
    return hints
```

N2 阈值按 §2；`content_chars` 与 Kotlin 一致（去标点空白，但**保留汉字/字母/数字**）；比率按千字算 `count * 1000 / chars`。

private suspend fun novel(dream: Dream, run: Run, idea: String?, onProgress: ...): GenerationResult {
    val raw = dream.rawTranscript.orEmpty()
    val base = dream.faithfulText?.takeIf { it.isNotBlank() } ?: raw

    // PLAN
    emit(GenerationKind.NOVEL, GenerationStage.PLAN, 1, onProgress)
    val planCall = retryingNovel(run, GenerationStage.PLAN) { callPlan(run, raw, base, idea) }
    val plan = planCall ?: return novelFailure(GenerationStage.PLAN, ...)
    val planGate = NovelGates.checkPlan(raw, plan.plan)
    if (planGate.failed.isNotEmpty()) {
        store.recordJob(job(run.dreamId, GenerationStage.PLAN, 1, error = planGate.summary()))
        return gateFailure(GenerationKind.NOVEL, GenerationStage.PLAN, planGate)
    }

    // DRAFT
    emit(GenerationKind.NOVEL, GenerationStage.DRAFT, 2, onProgress)
    val draft = retryingNovel(run, GenerationStage.DRAFT) { callDraft(run, raw, base, plan, idea) }
        ?: return novelFailure(...)

    // POLISH
    emit(GenerationKind.NOVEL, GenerationStage.POLISH, 3, onProgress)
    val polished = retryingNovel(run, GenerationStage.POLISH) { callPolish(run, draft, plan) }
    val text = polished?.body ?: draft
    val n2 = NovelGates.checkText(text)
    if (n2.failed.isNotEmpty()) return gateFailure(GenerationKind.NOVEL, GenerationStage.POLISH, n2)

    return commitNovel(run, GenerationStage.POLISH, text, notes = polished?.notes ?: fallbackNotes(plan), promptHash = NovelPrompts(context).bundleSha8())
}
```

细节：
- 每段调用前 `emit`，调用后无论成败都 `store.recordJob(...)`（stage=该段，attemptNo=1/2，error=null 或摘要）。
- `Thinking` 用 `run.novelThinking`（现有逻辑）；三段都 `jsonMode = true`（PLAN/POLISH 必须，DRAFT 要正文可关，但保持 JSON 更稳：DRAFT 输出 `{"body":"..."}`）。
- 字数由 POLISH 的提示词约束；N2 只校验最终 body。
